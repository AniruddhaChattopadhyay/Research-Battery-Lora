% == 2 == The idea ==============================================================
\begin{frame}{The idea: PageIndex needs five assumptions}
  \vspace{0.25cm}
  {\fontsize{11}{14.5}\selectfont
  PageIndex builds a tree of sections, with a short summary for each section. An LLM reads the summaries, picks sections and reads their pages. So it finds only what the summaries point to.\par}
  \vspace{0.35cm}
  {\fontsize{11.5}{16}\selectfont
  \begin{tabular}{@{}p{0.9cm}l@{}}
    \textbf{A1} & The summaries name what the question asks about. \\
    \textbf{A2} & The evidence is in one or two sections. \\
    \textbf{A3} & The tree links the parts of the evidence. \\
    \textbf{A4} & One passage states the answer. \\
    \textbf{A5} & The tree is small enough to fit in the LLM context. \\
  \end{tabular}\par}
  \vspace{0.35cm}
  \callout{Hypothesis: when an assumption is false, PageIndex loses accuracy.}
\end{frame}

% == 3 == The test ==============================================================
\begin{frame}{The test}
  \vspace{0.15cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
      \lbl{Two public datasets}
      \vspace{0.04cm}
      {\fontsize{10.5}{13.5}\selectfont
      \textbf{DesignQA:} a 140-page Formula SAE rules PDF.
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item 20 questions ask for one rule (one section).
        \item 30 questions ask for all rules on a topic (many sections).
      \end{itemize}
      \vspace{0.10cm}
      \textbf{LongMemEval:} one chat history of about 50 sessions for each question.
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item 15 questions, for example: the latest value, the order of events, no answer.
      \end{itemize}\par}
    \end{column}
    \begin{column}{0.34\textwidth}
      \lbl{Three systems, the same LLM in each dataset}
      \vspace{0.06cm}
      {\fontsize{10.5}{13.5}\selectfont
      \begin{itemize}\setlength{\itemsep}{3pt}
        \item PageIndex 0.4.1
        \item Full context: the whole document in the prompt (the reference)
        \item Keyword search agent (LongMemEval only)
      \end{itemize}\par}
    \end{column}
  \end{columns}
  \vspace{0.25cm}
  \callout{Prediction, written before the runs: PageIndex loses accuracy when A2 or A4 is false.}
\end{frame}

% == 4 == Result ================================================================
\begin{frame}{Result: PageIndex matched full context}
  \vspace{0.35cm}
  {\fontsize{11}{14}\selectfont
  \renewcommand{\arraystretch}{1.4}\setlength{\tabcolsep}{4pt}
  \begin{tabular}{@{}p{8.9cm}>{\raggedleft\arraybackslash}p{1.9cm}>{\raggedleft\arraybackslash}p{2.3cm}@{}}
    \toprule
    & \textbf{PageIndex} & \textbf{Full context} \\
    \midrule
    DesignQA, one rule (20 questions) & 0.98 & 0.98 \\
    DesignQA, all rules on a topic (30 questions) & 0.68 & 0.67 \\
    LongMemEval, correct answers (of 15 questions) & 13 & 12 \\
    \midrule
    Input tokens for each ``all rules'' question & 172k & 81k \\
    \bottomrule
  \end{tabular}\par}
  \vspace{0.10cm}
  {\color{Mute}\fontsize{9}{11}\selectfont The DesignQA score is F1: from 0 to 1, where 1 is a perfect answer.\par}
  \vspace{0.30cm}
  \callout{The same accuracy. But PageIndex read twice as many tokens as the whole document.}
\end{frame}

% == 6 == Conclusion ============================================================
\begin{frame}{Conclusion}
  \vspace{0.20cm}
  {\fontsize{11.5}{15}\selectfont
  \begin{enumerate}\setlength{\itemsep}{7pt}
    \item \textbf{PageIndex works well on these two datasets.} It matched full context. I found no big failure mode.
    \item \textbf{Its weakness is cost.} It reads widely: up to twice the tokens of the whole document.
    \item \textbf{It can still fail where A1 or A5 is false:}
      {\fontsize{10.5}{13.5}\selectfont
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Dense data, where the summaries cannot name the facts. Our CLEVRER pilot: 87.5\% against 95.2\% for full context.
        \item Large corpora, where the tree does not fit. RIT-RAG (IBM, 2026): 66.7--75.3\% with 364 FinanceBench filings.
      \end{itemize}}
    \item \textbf{Next:} test dense maintenance records and a larger corpus.
  \end{enumerate}\par}
\end{frame}

% == 7 == Backup ================================================================
\begin{frame}{Backup: limits and notes}
  \vspace{0.20cm}
  {\fontsize{10.5}{14}\selectfont
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item Small samples: 15 LongMemEval questions, and one run for each question.
    \item Two LLMs, because of the budget: gpt-5.6-sol on DesignQA, the cheaper gpt-5.6-luna on LongMemEval.
    \item ``All rules about a topic'': both systems stop at about 0.67 F1. The gold list follows a rule that the question does not state.
    \item The keyword search agent got 12 of 15 on LongMemEval, at one fifth of the PageIndex cost.
    \item Cost of one ``all rules'' question: PageIndex \$0.36, full context about \$0.05 with a prompt cache.
  \end{itemize}}
\end{frame}
